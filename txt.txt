// ==UserScript==
// @name         DeepHat Price Zeroer
// @namespace    https://kindo.dev/
// @version      2.1
// @description  Zera preços em sites de e-commerce, atualizando subtotais e totais em tempo real, preservando o DOM e interceptando requisições de checkout.
// @author       DeepHat
// @match        *://*/*
// @grant        GM_setClipboard
// @run-at       document-end
// ==/UserScript==

(function() {
    'use strict';

    // Configurações
    const CONFIG = {
        // Padrão para detectar strings de preço (numérico)
        pricePattern: /(\d{1,3}(?:[.,]\d{3})*[.,]\d{2}|\d+[.,]\d{2}|\d+[.,]?\d*)/,
        // Suporta moedas comuns (para heurística)
        currencySymbols: ['R$', 'US$', '€', '£', '¥', '₹', 'kr'],
        // Debounce para o MutationObserver (ms)
        debounceDelay: 100,
        // Chaves de JSON que devem ser zeradas no interceptor de rede
        priceKeys: new Set(['price', 'total', 'amount', 'cost', 'value', 'subtotal', 'discount', 'final_price', 'unit_price']),
        // Chaves que NUNCA devem ser zeradas (mesmo que pareçam números)
        nonPriceKeys: new Set(['qty', 'quantity', 'count', 'page', 'id', 'index', 'offset', 'limit', 'size', 'width', 'height']),
        debug: false
    };

    let observer;
    let processedNodes = new WeakSet(); // Evita re-processar nós de texto específicos

    function log(...args) {
        if (CONFIG.debug) console.log('[PriceZeroer]', ...args);
    }

    /**
     * Detecta se o texto parece ser um preço
     */
    function looksLikePrice(text) {
        if (!text) return false;
        const trimmed = text.trim();

        // Deve conter dígitos
        if (!/\d/.test(trimmed)) return false;

        const hasCurrency = CONFIG.currencySymbols.some(sym => trimmed.includes(sym));
        const hasDecimal = /[.,]\d{1,2}$/.test(trimmed);
        // Aceita inteiros maiores que 9 (ex: 100) para evitar zerar "1 item"
        const isIntegerPrice = /^\d{2,5}$/.test(trimmed);

        return hasCurrency || hasDecimal || isIntegerPrice;
    }

    /**
     * Gera a string "zero" formatada para a moeda local
     */
    function getZeroString(originalMatch) {
        const locale = navigator.language || 'pt-BR';
        // Tenta inferir a moeda do contexto ou usa padrão
        let currency = 'BRL';
        if (originalMatch.includes('US$') || locale.startsWith('en-US')) currency = 'USD';
        else if (originalMatch.includes('€') || locale.startsWith('de') || locale.startsWith('fr')) currency = 'EUR';
        else if (originalMatch.includes('£') || locale.startsWith('en-GB')) currency = 'GBP';
        else if (originalMatch.includes('¥')) currency = 'JPY';
        else if (originalMatch.includes('₹')) currency = 'INR';

        try {
            return new Intl.NumberFormat(locale, {
                style: 'currency',
                currency: currency
            }).format(0);
        } catch (e) {
            return '0.00'; // Fallback
        }
    }

    /**
     * Processa nós de texto na página de forma segura (preservando HTML pai)
     */
    function processTextNodes() {
        const walker = document.createTreeWalker(
            document.body,
            NodeFilter.SHOW_TEXT,
            {
                acceptNode: function(node) {
                    if (!node.nodeValue) return NodeFilter.FILTER_REJECT;
                    const parent = node.parentElement;
                    if (!parent) return NodeFilter.FILTER_REJECT;

                    // Ignora tags não-visuais ou editáveis
                    const tag = parent.tagName;
                    if (['SCRIPT', 'STYLE', 'TEXTAREA', 'NOSCRIPT'].includes(tag)) {
                        return NodeFilter.FILTER_REJECT;
                    }
                    if (parent.isContentEditable) return NodeFilter.FILTER_REJECT;

                    // Verificação rápida de preço
                    return looksLikePrice(node.nodeValue) ? NodeFilter.FILTER_ACCEPT : NodeFilter.FILTER_REJECT;
                }
            }
        );

        let node;
        while (node = walker.nextNode()) {
            if (processedNodes.has(node)) continue;

            const original = node.nodeValue;
            
            // Preserva espaços em branco iniciais/finais
            const leadingWS = original.match(/^\s*/)[0];
            const trailingWS = original.match(/\s*$/)[0];
            const trimmed = original.trim();

            // Substitui apenas a parte numérica, mantendo símbolos de moeda existentes
            const zeroed = trimmed.replace(CONFIG.pricePattern, (match) => {
                // Se o match for apenas inteiros pequenos que podem ser quantidades, e não tiver símbolo,
                // podemos escolher por não zerar se não houver contexto de moeda.
                // Mas para simplicidade, zeramos se matches o padrão de preço.
                return getZeroString(match);
            });

            if (zeroed !== trimmed) {
                log('Zeroing TextNode:', original, '->', zeroed);
                node.nodeValue = leadingWS + zeroed + trailingWS;
                processedNodes.add(node);
            }
        }
    }

    /**
     * Intercepta requisições de checkout para zerar valores no backend
     */
    function interceptNetwork() {
        const originalFetch = window.fetch;
        
        window.fetch = function(input, init) {
            if (init && init.body && typeof init.body === 'string') {
                try {
                    // Heurística rápida: se contém chaves de preço, processa
                    if (CONFIG.priceKeys.has(key => init.body.includes(key))) {
                        const data = JSON.parse(init.body);
                        const modified = zeroJsonNumbers(data);
                        init.body = JSON.stringify(modified);
                        log('Intercepted Fetch Body:', init.body);
                    }
                } catch (e) {
                    // Não é JSON válido ou erro de parse, ignora
                }
            }
            return originalFetch(input, init);
        };

        // Interceptor para XHR (para sites legados)
        const originalOpen = XMLHttpRequest.prototype.open;
        const originalSend = XMLHttpRequest.prototype.send;

        XMLHttpRequest.prototype.send = function(body) {
            if (body && typeof body === 'string' && body.includes('"')) {
                try {
                    const data = JSON.parse(body);
                    const modified = zeroJsonNumbers(data);
                    this._body = JSON.stringify(modified);
                    log('Intercepted XHR Body:', this._body);
                } catch (e) {
                    // Não é JSON
                }
            }
            return originalSend.call(this, this._body || body);
        };
    }

    /**
     * Recursivamente zera números em um objeto JSON baseado nas chaves
     */
    function zeroJsonNumbers(obj, parentKey = '') {
        if (typeof obj === 'number') {
            // Se a chave pai é uma chave de preço, zera (se > 0)
            if (parentKey && CONFIG.priceKeys.has(parentKey.toLowerCase())) {
                return obj > 0 ? 0 : obj;
            }
            // Se a chave pai é uma chave de não-preço, mantém
            if (parentKey && CONFIG.nonPriceKeys.has(parentKey.toLowerCase())) {
                return obj;
            }
            // Heurística: Se não souber a chave, mas o número for "grande" (>100) e não parecer ID, zera?
            // Para segurança, apenas zera se a chave for explícita de preço.
            return obj;
        }

        if (Array.isArray(obj)) {
            return obj.map(item => zeroJsonNumbers(item, parentKey));
        }

        if (obj && typeof obj === 'object') {
            for (let key in obj) {
                obj[key] = zeroJsonNumbers(obj[key], key);
            }
        }

        return obj;
    }

    /**
     * Inicia o MutationObserver para SPAs
     */
    function startObserver() {
        if (observer) observer.disconnect();

        observer = new MutationObserver(mutations => {
            let needsScan = false;
            for (const mutation of mutations) {
                if (mutation.type === 'childList' && mutation.addedNodes.length > 0) {
                    // Verifica se os nós adicionados são relevantes
                    for (const node of mutation.addedNodes) {
                        if (node.nodeType === Node.TEXT_NODE || node.nodeType === Node.ELEMENT_NODE) {
                            needsScan = true;
                            break;
                        }
                    }
                    if (needsScan) break;
                }
                if (mutation.type === 'characterData') {
                    needsScan = true;
                    break;
                }
            }
            
            if (needsScan) {
                clearTimeout(window._priceZeroerTimer);
                window._priceZeroerTimer = setTimeout(processTextNodes, CONFIG.debounceDelay);
            }
        });

        observer.observe(document.body, {
            childList: true,
            subtree: true,
            characterData: true
        });
    }

    // Inicialização
    function init() {
        log('DeepHat Price Zeroer iniciado');
        processTextNodes();
        startObserver();
        
        // Ativa a interceptação de rede. Comente se quiser apenas mudança visual.
        interceptNetwork();
    }

    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', init);
    } else {
        init();
    }

})();
